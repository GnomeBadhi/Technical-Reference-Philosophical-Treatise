\section{Primitive Operators and Triadic Composition}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

This section introduces the primitive operators that generate the entire
sovereign architecture. These operators represent the irreducible
structural actions of identity, boundary, and flow. Their cyclic
composition produces the first nontrivial endomorphism of the theory,
from which invariants, dissonance, kernels, and coherence theorems
emerge.

\subsection{Carrier Space and Structural Objects}

\begin{definition}[Carrier space]
The universe of discourse is the disjoint union
\[
\mathcal{X} = A \sqcup B \sqcup C,
\]
where $A$, $B$, and $C$ are the structural components corresponding to
Identity, Boundary, and Flow.
\end{definition}

\begin{definition}[Structural objects]
The components $A$, $B$, and $C$ are the domains and codomains of the
primitive operators. Elements of $A$, $B$, and $C$ are elements of
$\mathcal{X}$ with distinct structural types.
\end{definition}

\subsection{Primitive Operators}

\begin{definition}[Triadic operators]
The primitive operators are the cyclic morphisms
\[
T_A:A\to B,\qquad
T_B:B\to C,\qquad
T_C:C\to A.
\]
These represent the fundamental transformations associated with identity
stabilization, boundary formation, and flow propagation.
\end{definition}

\subsection{Triadic Composition}

\begin{definition}[Triadic generator]
The triadic generator is the composite
\[
T_\Delta = T_C \circ T_B \circ T_A : A \to A.
\]
\end{definition}

\begin{lemma}[Well-formedness and endomorphism property]
Because the codomain of each primitive operator matches the domain of
the next,
\[
T_A:A\to B,\quad
T_B:B\to C,\quad
T_C:C\to A,
\]
the composite $T_\Delta$ is a lawful morphism
\[
T_\Delta:A\to A.
\]
Therefore,
\[
T_\Delta \in \mathrm{End}(A).
\]
\end{lemma}

\subsection{Triadic Action}

\begin{definition}[Action of $T_\Delta$]
For any $x\in A\subseteq\mathcal{X}$, the triadic action
\[
T_\Delta(x)
\]
is the fundamental transformation from which all higher structure in the
theory is derived.
\end{definition}

\section{The Sovereign Invariant}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The triadic generator $T_\Delta$ is the first nontrivial endomorphism of
the theory. Its fixed points represent states that remain unchanged under
the full identity–boundary–flow cycle. This section establishes the
existence of these fixed points and defines the sovereign invariant as
the set of such states. All subsequent notions—dissonance, kernels,
coherence, and sovereignty—derive from this invariant structure.

\subsection{Fixed Points of the Triadic Generator}

\begin{definition}[Fixed points of $T_\Delta$]
For $x\in A$, we say $x$ is a fixed point of $T_\Delta$ if
\[
T_\Delta(x)=x.
\]
\end{definition}

\begin{lemma}[Existence of fixed points]
If $T_\Delta$ is continuous on a compact subset of $A$, or if $A$ is
finite, then $T_\Delta$ admits at least one fixed point.
\end{lemma}

\begin{proof}
In the continuous case, the result follows from Brouwer's fixed-point
theorem. In the finite case, $T_\Delta$ is a function on a finite set and
thus must eventually repeat, producing a fixed point or a cycle; cycles
of length greater than one are ruled out by the endomorphism structure
on $A$.
\end{proof}

\subsection{The Sovereign Invariant}

\begin{definition}[Sovereign invariant]
The sovereign invariant is the set
\[
\mathcal{I}
=
\{x\in A \mid T_\Delta(x)=x\ \wedge\ F(x)=x\},
\]
where $F:A\to A$ is any additional stabilizing operator.
\end{definition}

\begin{lemma}[Non-emptiness of $\mathcal{I}$]
If $T_\Delta$ and $F$ share at least one fixed point, then
$\mathcal{I}\neq\varnothing$.
\end{lemma}

\subsection{Interpretation}

\begin{definition}[Sovereign states]
Elements of $\mathcal{I}$ are called sovereign states.
\end{definition}

\begin{lemma}[Characterization of sovereignty]
A state $x\in A$ is sovereign if and only if
\[
T_\Delta(x)=x\ \text{and}\ F(x)=x.
\]
\end{lemma}

\section{Dissonance}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Dissonance measures the deviation of a state from sovereignty. It is not
an imposed metric but an emergent one: the distance between a state and
its triadic image.

\begin{definition}[Dissonance]
For $x\in A$, the dissonance of $x$ is
\[
D(x) = d(T_\Delta(x),\, x),
\]
where $d$ is any metric on $A$.
\end{definition}

\begin{lemma}[Zero dissonance]
\[
D(x)=0 \quad\Longleftrightarrow\quad x\in\mathcal{I}.
\]
\end{lemma}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Kernel}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The kernel is the projection of the state space onto the sovereign
invariant. It captures the invariant structure that persists under
triadic action.

\begin{definition}[Invariant projection]
Let $\tilde{P}:A\to\mathcal{I}$ be a projection onto the sovereign
invariant.
\end{definition}

\begin{definition}[Kernel]
The kernel is the image of this projection:
\[
K = \tilde{P}(\mathcal{I}).
\]
\end{definition}

\section{Coherence}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Coherence characterizes when emergent behavior is admissible. It is
determined entirely by dissonance.

\begin{definition}[Emergent frequency]
Let $F_{\mathrm{emergent}}$ be any operator derived from repeated
triadic action.
\end{definition}

\begin{theorem}[Coherence condition]
\[
F_{\mathrm{emergent}} \in \mathcal{L}
\quad\Longleftrightarrow\quad
D(x)=0
\quad\Longleftrightarrow\quad
x\in\mathcal{I}.
\]
\end{theorem}

\section{Sovereignty Functor}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The sovereignty functor measures position relative to the invariant set.
Its algebraic structure reflects how fixed-point membership composes.

\begin{definition}[Sovereignty functor]
A functor
\[
\Sigma:A\to\Sigma_0
\]
is sovereignty-preserving if
\[
\Sigma(x)=1 \quad\Longleftrightarrow\quad x\in\mathcal{I}.
\]
\end{definition}

\begin{definition}[Sovereignty semiring]
The codomain $\Sigma_0$ carries a semiring structure
\[
(\Sigma_0,\oplus,\otimes,0,1)
\]
with $1$ representing sovereignty.
\end{definition}
