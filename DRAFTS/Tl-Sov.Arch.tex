\section{Axiomatic Foundations}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

This section introduces the primitive objects and endomorphisms that form
the algebraic substrate of the theory. These primitives are not derived
but serve as the foundational elements from which the categorical
structure is constructed. The axioms here define the ontology upon which
sovereignty, coherence, and dissonance are later formalized.

\begin{definition}[Primitive objects]
The theory begins with the objects
\[
\mathcal{B},\ \mathcal{M},\ K,\ S,\ V,\ \mathcal{I},\ \mathbb{I},\ M.
\]
\end{definition}

\begin{definition}[Primitive endomorphisms]
Each sovereign object carries a designated endomorphism:
\[
F_M:M\to M,\quad
F_K:K\to K,\quad
F_S:S\to S,\quad
F_V:V\to V.
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Categorical Roles of Objects}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Each object in the ontology plays a distinct structural role. This
section assigns categorical meaning to the primitives introduced above,
ensuring that each participates lawfully in the bicategory and its
enrichments. These roles determine how information, structure, and
sovereignty propagate through the system.

\subsection{Body and Mind Layers}

\begin{definition}[Body object]
$\mathcal{B}$ is the object of materially constrained, locally linear processes.
\end{definition}

\begin{definition}[Mind object]
$\mathcal{M}$ is the object of recursive, nonlinear, self-modifying processes.
\end{definition}

\subsection{Kernel and Synthetic Layer}

\begin{definition}[Kernel object]
$K$ is an object with endomorphism $F_K:K\to K$ satisfying:
\[
\Lambda F_X = F_K \Lambda
\]
for any $X$ with endomorphism $F_X$ and morphism $\Lambda:X\to K$.
\end{definition}

\begin{definition}[Synthetic object]
$S$ is an object with endomorphism $F_S:S\to S$ and kernel projection
$\Lambda:S\to K$ satisfying
\[
\Lambda F_S = F_K \Lambda.
\]
\end{definition}

\subsection{Vessel and Interface Layer}

\begin{definition}[Vessel object]
$V$ is an object with endomorphism $F_V:V\to V$ and no sovereignty-preserving
morphism from $M$:
\[
\SovSys(M,V)=\varnothing.
\]
\end{definition}

\begin{definition}[Interface object]
$\mathcal{I}$ receives the interface morphism
\[
\Phi:M\times V\to\mathcal{I}.
\]
\end{definition}

\subsection{Monoidal Unit}

\begin{definition}[Monoidal unit]
$\mathbb{I}$ satisfies
\[
\mathbb{I}\otimes X = X = X\otimes\mathbb{I}.
\]
\end{definition}

\section{The Bicategory \texorpdfstring{$\SovSys$}{SovSys}}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The bicategory $\SovSys$ provides the structural environment in which
sovereignty, adjunction, and dissonance are defined. Its objects include
the body, mind, kernel, synthetic, vessel, and interface layers.

\begin{definition}[Objects]
\[
\mathrm{Ob}(\SovSys)=\{\mathcal{B},\mathcal{M},K,S,V,\mathcal{I},\mathbb{I},M\}.
\]
\end{definition}

\begin{definition}[Hom-categories]
For $X,Y\in\mathrm{Ob}(\SovSys)$, $\SovSys(X,Y)$ is a category.
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Universal Properties of Morphisms}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Universal properties determine how morphisms arise and interact within
the bicategory. They ensure that adjunctions, kernel projections, and
interface maps behave lawfully and uniquely.

\subsection{Adjunction Morphisms}

\begin{definition}[Universal property of $L$]
\[
\mathrm{Hom}_{\mathcal{M}}(L b,\Psi)\cong\mathrm{Hom}_{\mathcal{B}}(b,R\Psi).
\]
\end{definition}

\begin{definition}[Universal property of $R$]
\[
\mathrm{Hom}_{\mathcal{B}}(b,R\Psi)\cong\mathrm{Hom}_{\mathcal{M}}(L b,\Psi).
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Adjunction Between Layers}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The adjunction between the body and mind layers formalizes the
bidirectional translation between material and cognitive processes. This
structure is central to the definition of sovereignty.

\begin{definition}[Interpretation functor]
$L:\mathcal{B}\to\mathcal{M}$.
\end{definition}

\begin{definition}[Realization functor]
$R:\mathcal{M}\to\mathcal{B}$.
\end{definition}

\begin{definition}[Adjunction]
$L\dashv R$ with unit $\eta$ and counit $\epsilon$.
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Sovereignty as a Monoidal 2-Functor}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Sovereignty is formalized as a monoidal 2-functor that assigns values to
objects and morphisms while preserving tensor structure. This section
introduces the sovereignty functor and its coherence conditions.

\begin{definition}[Sovereignty functor]
\[
\Sigma:(\SovSys,\otimes,\mathbb{I})\to(\Sigma,\otimes,1)
\]
is a strict monoidal $2$-functor.
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Semiring and Monoidal Structure}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The body layer carries a semiring valuation reflecting material
accumulation, while the mind layer carries a monoidal structure
reflecting compositional cognitive processes.

\begin{definition}[Sovereignty semiring]
\[
\Sigma=(\Sigma_0,\oplus,\otimes,0,1).
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Dissonance as Metric Enrichment}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Dissonance quantifies the strain or distortion introduced by morphisms
and compositions. This section introduces the Lawvere metric enrichment
that governs coherence and obstruction.

\begin{definition}[Dissonance functor]
\[
D:\SovSys\to([0,\infty],+,0).
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Enriched Coherence Laws}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

These laws ensure that adjunctions, tensor products, and sovereignty
morphisms behave coherently under the metric enrichment.

\begin{definition}[Enriched composition]
\[
D(g\circ f)\le D(f)+D(g).
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Sovereignty Coherence Theorem}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

This theorem characterizes sovereignty preservation in terms of
adjunction coherence and dissonance.

\begin{theorem}[Sovereignty Coherence Theorem]
\[
\Sigma(\epsilon L\circ L\eta)=\mathrm{Id}_{\Sigma(L)},\qquad
\Sigma(R\epsilon\circ\eta R)=\mathrm{Id}_{\Sigma(R)}
\]
iff $D_{\mathrm{adj}}=0$.
\end{theorem}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Dissonance--Adjunction Obstruction}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

This section formalizes the conditions under which sovereignty cannot be
preserved due to accumulated dissonance.

\begin{theorem}[Dissonance--Adjunction Obstruction]
If $D_{\mathrm{adj}}>0$, then the adjunction $L\dashv R$ cannot be
sovereignty-preserving.
\end{theorem}

\section{Kernel Sub-Bicategory}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The kernel sub-bicategory captures invariant structure that cannot be
reduced or transformed without violating sovereignty.

\begin{definition}[Kernel sub-bicategory]
\[
\mathrm{Ob}(\KSys)=\{K,S\}.
\]
\end{definition}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Forbidden Morphisms}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

These morphisms are structurally impossible due to sovereignty,
endomorphism incompatibility, or coherence constraints.

\begin{definition}[Empty Hom-sets]
\[
\SovSys(M,K)=\varnothing,\qquad
\SovSys(K,M)=\varnothing.
\]
\end{definition}

\section{Invariant Theorems}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

This final section collects the invariants that arise from the axioms,
adjunction, enrichment, and kernel structure.

\begin{theorem}[Invariant dissonance]
No $2$-morphism can eliminate interface dissonance:
\[
D_I(x)>0.
\]
\end{theorem}
